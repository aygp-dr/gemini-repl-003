* Git Hooks

** Pre-commit Hook

#+BEGIN_SRC bash :tangle .git/hooks/pre-commit :shebang #!/bin/sh
# Pre-commit hook for Gemini REPL project

# Add trailing newlines to text files
for file in $(git diff --cached --name-only); do
    if file "$file" | grep -q text; then
        if [ -f "$file" ] && [ "$(tail -c 1 "$file")" != "" ]; then
            echo >> "$file"
            git add "$file"
        fi
    fi
done

# Check for common issues
echo "Running pre-commit checks..."

# Check for TODO/FIXME in staged files
if git diff --cached | grep -E '^\+.*(TODO|FIXME)'; then
    echo "Warning: TODO or FIXME found in staged changes"
fi

exit 0
#+END_SRC
